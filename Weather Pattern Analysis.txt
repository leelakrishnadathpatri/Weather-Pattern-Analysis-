#include <stdio.h>

#define MAX_DAYS 100

int main()
{
    int n, i;
    float temperature[MAX_DAYS];
    float humidity[MAX_DAYS];
    float rainfall[MAX_DAYS];

    float totalTemp = 0, totalHumidity = 0;
    float totalRainfall = 0;
    float maxTemp, minTemp;
    int hotDays = 0, rainyDays = 0, coldDays = 0;

    printf("====================================\n");
    printf("       WEATHER PATTERN ANALYSIS\n");
    printf("====================================\n");

    printf("Enter number of days (1-%d): ", MAX_DAYS);
    if (scanf("%d", &n) != 1 || n < 1 || n > MAX_DAYS)
    {
        printf("Invalid number of days!\n");
        return 1;
    }

    for (i = 0; i < n; i++)
    {
        printf("\nDay %d\n", i + 1);

        printf("Enter temperature (Celsius): ");
        if (scanf("%f", &temperature[i]) != 1)
            return 1;

        printf("Enter humidity (%%): ");
        if (scanf("%f", &humidity[i]) != 1 ||
            humidity[i] < 0 || humidity[i] > 100)
        {
            printf("Invalid humidity value!\n");
            return 1;
        }

        printf("Enter rainfall (mm): ");
        if (scanf("%f", &rainfall[i]) != 1 ||
            rainfall[i] < 0)
        {
            printf("Invalid rainfall value!\n");
            return 1;
        }

        totalTemp += temperature[i];
        totalHumidity += humidity[i];
        totalRainfall += rainfall[i];

        if (i == 0)
        {
            maxTemp = temperature[i];
            minTemp = temperature[i];
        }

        if (temperature[i] > maxTemp)
            maxTemp = temperature[i];

        if (temperature[i] < minTemp)
            minTemp = temperature[i];

        if (temperature[i] >= 35)
            hotDays++;

        if (temperature[i] < 20)
            coldDays++;

        if (rainfall[i] > 0)
            rainyDays++;
    }

    printf("\n====================================\n");
    printf("          ANALYSIS REPORT\n");
    printf("====================================\n");

    printf("Number of days       : %d\n", n);
    printf("Average temperature  : %.2f C\n",
           totalTemp / n);
    printf("Highest temperature  : %.2f C\n", maxTemp);
    printf("Lowest temperature   : %.2f C\n", minTemp);
    printf("Average humidity     : %.2f %%\n",
           totalHumidity / n);
    printf("Total rainfall       : %.2f mm\n",
           totalRainfall);
    printf("Rainy days           : %d\n", rainyDays);
    printf("Hot days (>=35 C)    : %d\n", hotDays);
    printf("Cold days (<20 C)    : %d\n", coldDays);

    printf("\nWeather Pattern: ");

    if (rainyDays >= n / 2.0 && totalRainfall > 0)
        printf("Rainy Weather\n");
    else if (totalTemp / n >= 35)
        printf("Hot Weather\n");
    else if (totalTemp / n < 20)
        printf("Cold Weather\n");
    else
        printf("Moderate Weather\n");

    printf("====================================\n");

    return 0;
}
